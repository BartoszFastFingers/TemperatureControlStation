50ARM_DSP_ATTRIBUTE void arm_iir_lattice_init_f32(
51 arm_iir_lattice_instance_f32 * S,
52 uint16_t numStages,
53 float32_t * pkCoeffs,
54 float32_t * pvCoeffs,
55 float32_t * pState,
56 uint32_t blockSize)
57{
58 /* Assign filter taps */
59 S->numStages = numStages;
60
61 /* Assign reflection coefficient pointer */
62 S->pkCoeffs = pkCoeffs;
63
64 /* Assign ladder coefficient pointer */
65 S->pvCoeffs = pvCoeffs;
66
67 /* Clear state buffer and size is always blockSize + numStages */
68 memset(pState, 0, (numStages + blockSize) * sizeof(float32_t));
69
70 /* Assign state pointer */
71 S->pState = pState;
72}
