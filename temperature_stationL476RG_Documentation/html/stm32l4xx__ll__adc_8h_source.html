5954__STATIC_INLINE uint32_t LL_ADC_GetAnalogWDMonitChannels(const ADC_TypeDef *ADCx, uint32_t AWDy)
5955{
5956 const __IO uint32_t *preg = __ADC_PTR_REG_OFFSET(ADCx->CFGR,
5957 ((AWDy & ADC_AWD_CRX_REGOFFSET_MASK) >> ADC_AWD_CRX_REGOFFSET_POS)
5958 + ((AWDy & ADC_AWD_CR12_REGOFFSETGAP_MASK)
5959 * ADC_AWD_CR12_REGOFFSETGAP_VAL));
5960
5961 uint32_t analog_wd_monit_channels = (READ_BIT(*preg, AWDy) & AWDy & ADC_AWD_CR_ALL_CHANNEL_MASK);
5962
5963 /* If "analog_wd_monit_channels" == 0, then the selected AWD is disabled */
5964 /* (parameter value LL_ADC_AWD_DISABLE). */
5965 /* Else, the selected AWD is enabled and is monitoring a group of channels */
5966 /* or a single channel. */
5967 if (analog_wd_monit_channels != 0UL)
5968 {
5969 if (AWDy == LL_ADC_AWD1)
5970 {
5971 if ((analog_wd_monit_channels & ADC_CFGR_AWD1SGL) == 0UL)
5972 {
5973 /* AWD monitoring a group of channels */
5974 analog_wd_monit_channels = ((analog_wd_monit_channels
5975 | (ADC_AWD_CR23_CHANNEL_MASK)
5976 )
5977 & (~(ADC_CFGR_AWD1CH))
5978 );
5979 }
5980 else
5981 {
5982 /* AWD monitoring a single channel */
5983 analog_wd_monit_channels = (analog_wd_monit_channels
5984 | (ADC_AWD2CR_AWD2CH_0 << (analog_wd_monit_channels >> ADC_CFGR_AWD1CH_Pos))
5985 );
5986 }
5987 }
5988 else
5989 {
5990 if ((analog_wd_monit_channels & ADC_AWD_CR23_CHANNEL_MASK) == ADC_AWD_CR23_CHANNEL_MASK)
5991 {
5992 /* AWD monitoring a group of channels */
5993 analog_wd_monit_channels = (ADC_AWD_CR23_CHANNEL_MASK
5994 | ((ADC_CFGR_JAWD1EN | ADC_CFGR_AWD1EN))
5995 );
5996 }
5997 else
5998 {
5999 /* AWD monitoring a single channel */
6000 /* AWD monitoring a group of channels */
6001 analog_wd_monit_channels = (analog_wd_monit_channels
6002 | (ADC_CFGR_JAWD1EN | ADC_CFGR_AWD1EN | ADC_CFGR_AWD1SGL)
6003 | (__LL_ADC_CHANNEL_TO_DECIMAL_NB(analog_wd_monit_channels) << ADC_CFGR_AWD1CH_Pos)
6004 );
6005 }
6006 }
6007 }
6008
