76ARM_DSP_ATTRIBUTE void arm_welch_f64(
77 float64_t * pDst,
78 uint32_t blockSize)
79{
80 float64_t k = 2.0 / ((float64_t) blockSize);
81 float64_t w;
82
83 for(uint32_t i=0;i<blockSize;i++)
84 {
85 w = i * k - 1.0;
86 w = 1.0 - w * w;
87 pDst[i] = w;
88 }
89}
