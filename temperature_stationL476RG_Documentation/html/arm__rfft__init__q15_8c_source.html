365ARM_DSP_ATTRIBUTE arm_status arm_rfft_init_q15(
366 arm_rfft_instance_q15 * S,
367 uint32_t fftLenReal,
368 uint32_t ifftFlagR,
369 uint32_t bitReverseFlag)
370{
371 /* Initialise the default arm status */
372 arm_status status = ARM_MATH_ARGUMENT_ERROR;
373
374 /* Initialization of coef modifier depending on the FFT length */
375 switch (fftLenReal)
376 {
377 case 8192U:
378 status = arm_rfft_init_8192_q15( S,ifftFlagR,bitReverseFlag );
379 break;
380 case 4096U:
381 status = arm_rfft_init_4096_q15( S,ifftFlagR,bitReverseFlag );
382 break;
383 case 2048U:
384 status = arm_rfft_init_2048_q15( S,ifftFlagR,bitReverseFlag );
385 break;
386 case 1024U:
387 status = arm_rfft_init_1024_q15( S,ifftFlagR,bitReverseFlag );
388 break;
389 case 512U:
390 status = arm_rfft_init_512_q15( S,ifftFlagR,bitReverseFlag );
391 break;
392 case 256U:
393 status = arm_rfft_init_256_q15( S,ifftFlagR,bitReverseFlag );
394 break;
395 case 128U:
396 status = arm_rfft_init_128_q15( S,ifftFlagR,bitReverseFlag );
397 break;
398 case 64U:
399 status = arm_rfft_init_64_q15( S,ifftFlagR,bitReverseFlag );
400 break;
401 case 32U:
402 status = arm_rfft_init_32_q15( S,ifftFlagR,bitReverseFlag );
403 break;
404 default:
405 /* Reporting argument error if rfftSize is not valid value */
406 status = ARM_MATH_ARGUMENT_ERROR;
407 break;
408 }
409
410 /* return the status of RFFT Init function */
411 return (status);
412}
