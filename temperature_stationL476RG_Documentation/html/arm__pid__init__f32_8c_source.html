50ARM_DSP_ATTRIBUTE void arm_pid_init_f32(
51 arm_pid_instance_f32 * S,
52 int32_t resetStateFlag)
53{
54 /* Derived coefficient A0 */
55 S->A0 = S->Kp + S->Ki + S->Kd;
56
57 /* Derived coefficient A1 */
58 S->A1 = (-S->Kp) - ((float32_t) 2.0f * S->Kd);
59
60 /* Derived coefficient A2 */
61 S->A2 = S->Kd;
62
63 /* Check whether state needs reset or not */
64 if (resetStateFlag)
65 {
66 /* Reset state to zero, The size will be always 3 samples */
67 memset(S->state, 0, 3U * sizeof(float32_t));
68 }
69
70}
